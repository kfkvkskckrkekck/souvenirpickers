                      {payoutInfo.bank_account_last4 && (

                          <p className="text-xs text-gray-600 mb-1">Bank Account</p>
                          <p className="font-semibold text-gray-900 text-sm">•••• {payoutInfo.bank_account_last4}</p>
                        </div>
                      )}
                    </div>

                    {!payoutInfo.payouts_enabled && (
                      <button
                        type="button"
                        onClick={async () => {
                          try {
                            setSavingPayout(true);
                            const supabaseUrl = import.meta.env.VITE_SUPABASE_URL;
                            const { data: profileData } = await supabase
                              .from('profiles')
                              .select('email')
                              .eq('id', user?.id)
                              .single();

                            const response = await fetch(
                              `${supabaseUrl}/functions/v1/create-stripe-connect-account`,
                              {
                                method: 'POST',
                                headers: {
                                  'Authorization': `Bearer ${(await supabase.auth.getSession()).data.session?.access_token}`,
                                  'Content-Type': 'application/json',
                                },
                                body: JSON.stringify({
                                  email: profileData?.email || user?.email,
                                  country: payoutInfo.country || 'US',
                                  businessType: 'individual',
                                  returnUrl: window.location.href,
                                  refreshUrl: window.location.href,
                                }),
                              }
                            );

                            const result = await response.json();
                            if (result.onboardingUrl) {
                              window.location.href = result.onboardingUrl;
                            }
                          } catch (error: any) {
                            setPayoutError(error.message);
                          } finally {
                            setSavingPayout(false);
                          }
                        }}
                        disabled={savingPayout}
                        className="w-full bg-yellow-500 text-white py-3 px-6 rounded-lg font-semibold hover:bg-yellow-600 transition-all flex items-center justify-center gap-2 disabled:opacity-50"
                      >
                        {savingPayout ? (
                          <>
                            <Loader className="w-5 h-5 animate-spin" />
                            Loading...
                          </>
                        ) : (
                          <>
                            <ExternalLink className="w-5 h-5" />
                            Complete Stripe Setup

                      </button>
